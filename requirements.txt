# cryptography is a direct, module-scope import of api/source_connections.py
# (Fernet for source credentials encrypted at rest). It was previously present
# only transitively through pdfminer.six, so an environment that omitted
# pdfminer.six silently dropped the /solspire/sources router -- the app still
# booted, because api/key_routes.py swallows the ImportError. The gate2 backend
# observation then compared a route set missing four operations against
# production and could not distinguish an environment gap from a source
# divergence. Declared explicitly so the dependency is a contract, not a
# coincidence of the dependency graph.
cryptography

# pytest-asyncio drives the async test modules (e.g. the API→enterprise
# authorization boundary in tests/test_authority_api_enterprise_boundary.py).
# CI installs pytest but not this plugin, so an async test collected without it
# is never awaited and asserts nothing -- a test that cannot fail is not
# coverage. Declared explicitly so the async boundary is actually exercised.
pytest-asyncio

# jsonschema validates the ARK-01 signal envelope against its frozen canonical
# contract (schemas/arkana/signal/1.0/arkana-signal.schema.json). That schema was
# admitted to the CP10 allowlist but nothing pinned the producer to it, so
# `POST /api/arkana/signal/ingest` emitted objects that fail their own contract
# for ordinary model output (null confidence in the number-only map; candidates
# missing the required `status`). Declared explicitly so the conformance test
# exercises the real schema rather than a hand-rolled approximation.
jsonschema
